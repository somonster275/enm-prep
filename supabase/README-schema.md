# Export du schéma Supabase

`supabase/schema.sql` décrit toute la structure de la base (schéma public) : tables, contraintes, index,
RLS, fonctions, triggers et droits d'exécution. **Pas de données** : ce n'est pas une sauvegarde.

À ré-exporter après chaque migration, pour que le fichier reste à jour :

1. Coller la requête ci-dessous dans le SQL Editor de Supabase. Elle ne fait que lire.
   L'avertissement « RLS » de l'éditeur est une fausse alerte (les mots `create table` ne sont que du texte) :
   choisir l'option qui lance la requête sans changement.
2. Copier le contenu de la case de résultat dans `supabase/schema.sql` (garder l'en-tête).

Non inclus : buckets Storage et leurs règles (`/api/storage/init` recrée les buckets), schéma `auth`.

```sql
with
tabs as (
  select c.oid, c.relname, c.relrowsecurity
  from pg_class c join pg_namespace n on n.oid = c.relnamespace
  where n.nspname = 'public' and c.relkind in ('r', 'p')
    and not exists (select 1 from pg_depend d where d.objid = c.oid and d.deptype = 'e')
),
ext as (
  select 1 as ordre, e.extname as tri,
    format('create extension if not exists %I with schema %I;', e.extname, n.nspname) as sql
  from pg_extension e join pg_namespace n on n.oid = e.extnamespace
  where e.extname not in ('plpgsql')
),
tables as (
  select 2, t.relname,
    format(E'create ' || E'table if not exists public.%I (\n%s\n);', t.relname, string_agg(
      format('  %I %s%s%s', a.attname, format_type(a.atttypid, a.atttypmod),
        case when a.attidentity = 'a' then ' generated always as identity'
             when a.attidentity = 'd' then ' generated by default as identity'
             when a.attgenerated = 's' then format(' generated always as (%s) stored', pg_get_expr(ad.adbin, ad.adrelid))
             when ad.adbin is not null then ' default ' || pg_get_expr(ad.adbin, ad.adrelid) else '' end,
        case when a.attnotnull then ' not null' else '' end),
      E',\n' order by a.attnum))
  from tabs t
  join pg_attribute a on a.attrelid = t.oid and a.attnum > 0 and not a.attisdropped
  left join pg_attrdef ad on ad.adrelid = a.attrelid and ad.adnum = a.attnum
  group by t.relname
),
contraintes as (
  select case when co.contype = 'f' then 4 else 3 end, t.relname || co.conname,
    format('alter table public.%I add constraint %I %s;', t.relname, co.conname, pg_get_constraintdef(co.oid))
  from tabs t join pg_constraint co on co.conrelid = t.oid
  where co.contype in ('p', 'u', 'c', 'f', 'x')
),
index as (
  select 5, t.relname || i.relname, pg_get_indexdef(ix.indexrelid) || ';'
  from tabs t
  join pg_index ix on ix.indrelid = t.oid
  join pg_class i on i.oid = ix.indexrelid
  where not exists (select 1 from pg_constraint co where co.conindid = ix.indexrelid)
),
rls as (
  select 6, t.relname, format('alter table public.%I enable row level security;', t.relname)
  from tabs t where t.relrowsecurity
),
fonctions as (
  select 7, p.proname || p.oid, pg_get_functiondef(p.oid) || ';'
  from pg_proc p join pg_namespace n on n.oid = p.pronamespace
  where n.nspname = 'public' and p.prokind in ('f', 'p')
    and not exists (select 1 from pg_depend d where d.objid = p.oid and d.deptype = 'e')
),
politiques as (
  select 8, p.tablename || p.policyname,
    format('create policy %I on public.%I as %s for %s to %s%s%s;', p.policyname, p.tablename, lower(p.permissive), lower(p.cmd),
      array_to_string(p.roles, ', '),
      case when p.qual is not null then format(' using (%s)', p.qual) else '' end,
      case when p.with_check is not null then format(' with check (%s)', p.with_check) else '' end)
  from pg_policies p where p.schemaname = 'public'
),
declencheurs as (
  select 9, c.relname || tg.tgname, pg_get_triggerdef(tg.oid) || ';'
  from pg_trigger tg
  join pg_class c on c.oid = tg.tgrelid
  join pg_namespace n on n.oid = c.relnamespace
  join pg_proc p on p.oid = tg.tgfoid
  join pg_namespace pn on pn.oid = p.pronamespace
  where not tg.tgisinternal and (n.nspname = 'public' or pn.nspname = 'public')
),
retraits as (
  -- Sans ce retrait, une base reconstruite rouvrirait à tous les fonctions dont l'exécution a été restreinte.
  select 10, p.proname || p.oid || '0',
    format('revoke all on function %s from public, anon, authenticated;', p.oid::regprocedure)
  from pg_proc p join pg_namespace n on n.oid = p.pronamespace
  where n.nspname = 'public' and p.proacl is not null
    and not exists (select 1 from pg_depend d where d.objid = p.oid and d.deptype = 'e')
),
droits_fonctions as (
  select 10, p.proname || p.oid || '1' || r.rolname,
    format('grant execute on function %s to %I;', p.oid::regprocedure, r.rolname)
  from pg_proc p join pg_namespace n on n.oid = p.pronamespace
  cross join lateral aclexplode(p.proacl) acl
  join pg_roles r on r.oid = acl.grantee
  where n.nspname = 'public' and p.proacl is not null and acl.privilege_type = 'EXECUTE'
    and r.rolname in ('anon', 'authenticated', 'service_role')
    and not exists (select 1 from pg_depend d where d.objid = p.oid and d.deptype = 'e')
),
tout as (
  select * from ext union all select * from tables union all select * from contraintes
  union all select * from index union all select * from rls union all select * from fonctions
  union all select * from politiques union all select * from declencheurs union all select * from retraits union all select * from droits_fonctions
)
select string_agg(sql, E'\n\n' order by ordre, tri) as schema from tout;
```
